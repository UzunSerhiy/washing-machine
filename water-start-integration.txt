def build_calibration_session_response(
    service: CalibrationSessionService,
) -> CalibrationSessionResponse:
    session = service.get_current()

    return CalibrationSessionResponse(
        active=True,
        material_profile_id=session.material_profile_id,
        position_turns=session.position_turns,
        running=session.player.running,
        paused=session.player.paused,
        direction=session.player.direction,
        frequency_hz=session.player.frequency_hz,
        washing_stop_turns=session.machine_calibration.washing_stop_turns,
        material_start_turns=session.machine_calibration.material_start_turns,
        material_end_turns=session.winding_calibration.material_end_turns,
        brush_running=session.player.brush_running,
        brush_frequency_hz=session.player.brush_frequency_hz,
        stage=session.stage,
    )


def stop_auto_before_mode_change():
    auto_service = get_auto_session_service()

    if auto_service.get_status()["status"] == "FAULT":
        raise HTTPException(
            status_code=409,
            detail="AUTO controller is in FAULT state",
        )

    if auto_service.get_status()["status"] != "IDLE":
        try:
            auto_service.stop()
        except (ValueError, RuntimeError) as exc:
            raise HTTPException(status_code=409, detail=str(exc)) from exc


async def get_enabled_material_profile(
    material_profile_id: int,
) -> MaterialProfile:
    async with AsyncSessionLocal() as session:
        result = await session.execute(
            select(MaterialProfile).where(
                MaterialProfile.id == material_profile_id,
                MaterialProfile.is_enabled.is_(True),
            )
        )

        profile = result.scalar_one_or_none()

    if profile is None:
        raise HTTPException(status_code=404, detail="Material profile not found")

    return profile


@router.get("", response_model=MachineResponse)
async def get_machine():
    async with AsyncSessionLocal() as session:
        result = await session.execute(
            select(Machine).options(selectinload(Machine.drives)).where(Machine.id == 1)
        )

        machine = result.scalar_one_or_none()

    if machine is None:
        raise HTTPException(
            status_code=404,
            detail="Machine not found",
        )

    drive_service = get_drive_service()

    drives = []

    for db_drive in machine.drives:
        if not db_drive.is_enabled:
            continue

        drive = drive_service.get_drive(db_drive.id)

        drives.append(
            MachineDriveResponse(
                id=db_drive.id,
                name=db_drive.name,
                address=db_drive.address,
                type=db_drive.type,
                position=db_drive.position,
                enabled=db_drive.is_enabled,
                status=drive.get_status(),
                frequency_hz=drive.get_frequency(),
                fault=drive.get_fault(),
            )
        )

    return MachineResponse(
        id=machine.id,
        name=machine.name,
        description=machine.description,
        is_active=machine.is_active,
        drives=drives,
    )

@router.post("/calibration/complete")
async def complete_calibration():
    service = get_calibration_session_service()

    try:
        calibration_session = service.complete()
    except (ValueError, RuntimeError) as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc

    async with AsyncSessionLocal() as db:
        machine_result = await db.execute(
            select(Machine).where(Machine.is_active.is_(True))
        )
        machine = machine_result.scalars().first()

        if machine is None:
            raise HTTPException(
                status_code=404,
                detail="Active machine not found",
            )

        repository = CalibrationRepository()

        try:
            await repository.save(
                db=db,
                machine_id=machine.id,
                calibration_session=calibration_session,
            )

            await db.commit()
            service.finish()

        except (ValueError, RuntimeError) as exc:
            await db.rollback()
            raise HTTPException(
                status_code=400,
                detail=str(exc),
            ) from exc

    operation_service = get_machine_operation_service()
    operation_service.set_manual_mode()

    return {
        "status": "completed",
        "machine_id": machine.id,
        "material_profile_id": calibration_session.material_profile_id,
    }

class IOOutputRequest(BaseModel):

@router.put("/io/{device}")
def set_io_output(device: str, request: IOOutputRequest):
    from app.main import machine_io_controller

    if device not in ("water", "light"):
        raise HTTPException(
            status_code=404,
            detail=f"Unknown IO device: {device}",
        )

    if machine_io_controller is None:
        raise HTTPException(
            status_code=503,
            detail="Machine IO controller is unavailable",
        )

    try:
        machine_io_controller.set_output(
            device=device,
            enabled=request.enabled,
        )

    except TimeoutError as exc:
        raise HTTPException(
            status_code=504,
            detail=str(exc),
        ) from exc

    except RuntimeError as exc:
        raise HTTPException(
            status_code=503,
            detail=str(exc),
        ) from exc

    return {
        "device": device,
        "requested": request.enabled,
        "status": "command_confirmed",
    }

} from "@/components/ui/select";

import { Slider } from "@/components/ui/slider";

function errorMessage(error: unknown): string {
  if (axios.isAxiosError(error)) {
    const detail = error.response?.data?.detail;
    return typeof detail === "string" ? detail : error.message;
  }

  return error instanceof Error ? error.message : "Unknown error";
}

function formatTurns(value: number | null | undefined) {
  return value == null ? "—" : value.toFixed(4);
}

const ROLLER_MAX_FREQUENCY_HZ = 10;

function percentToHz(percent: number): number {
  return (percent / 100) * ROLLER_MAX_FREQUENCY_HZ;
}

const BRUSH_MAX_FREQUENCY_HZ = 10;

function brushPercentToHz(percent: number): number {
  return (percent / 100) * BRUSH_MAX_FREQUENCY_HZ;
}

export default function CalibrationPage() {
  const [materialId, setMaterialId] = useState("");
  const [rollerSpeedPercent, setRollerSpeedPercent] = useState(10);
  const [brushSpeedPercent, setBrushSpeedPercent] = useState(30);
  const [error, setError] = useState<string | null>(null);
  const [message, setMessage] = useState<string | null>(null);
  const [positionConfirmed, setPositionConfirmed] = useState(false);

  const materialsQuery = useMaterialProfiles();
  const sessionQuery = useCalibrationSession();
  const homeQuery = useCalibrationHome();
  const sensorQuery = useRollerReference();
  const startHome = useStartCalibrationHome();

  const start = useStartCalibration();
  const jog = useJogCalibration();
  const action = useCalibrationAction();
  const mark = useMarkCalibrationPoint();
  const complete = useCompleteCalibration();
  const cancel = useCancelCalibration();

  const session = sessionQuery.data;
  const active = session?.active === true;
  const homingActive = homeQuery.data?.state === "searching";

  const startBrush = useStartCalibrationBrush();
  const stopBrush = useStopCalibrationBrush();

  const busy =
    start.isPending ||
    jog.isPending ||
    action.isPending ||
    mark.isPending ||
    complete.isPending ||
    cancel.isPending ||
    startBrush.isPending ||
    stopBrush.isPending ||
    startHome.isPending;

  const canComplete =
    active &&
    !busy &&
    !session?.running &&
    !session?.brush_running &&
    session?.material_start_turns != null &&
    session?.material_end_turns != null &&
    session?.washing_stop_turns != null;

  async function execute(fn: () => Promise<unknown>) {
    setError(null);
    setMessage(null);

    try {
      await fn();
    } catch (err) {
      setError(errorMessage(err));
    }
  }

  function handleStart() {
    const id = Number(materialId);

    if (!Number.isInteger(id) || id <= 0) {
      setError("Выбери материал");
      return;
    }

    execute(() => start.mutateAsync(id));
  }

  function handleJog(direction: "forward" | "reverse") {
    if (
      !Number.isFinite(rollerSpeedPercent) ||
      rollerSpeedPercent <= 0 ||
      rollerSpeedPercent > 100
    ) {
      setError("Скорость должна быть от 1 до 100%");
      return;
    }

    const frequencyHz = percentToHz(rollerSpeedPercent);

    execute(() =>
      jog.mutateAsync({
        direction,
        frequencyHz,
      }),
    );
  }

  function handleComplete() {
    execute(async () => {
      await complete.mutateAsync();
      setMessage("Калибровка сохранена в PostgreSQL");
    });
  }

  function handleCancel() {
    execute(() => cancel.mutateAsync());
  }

  const stage = session?.stage;

  const steps = [
    { key: "set_zero", label: "Установка нуля" },
    { key: "material_start", label: "Начало материала" },
    { key: "material_end", label: "Конец материала" },
    { key: "ready_to_unwind", label: "Автоматическая отмотка" },
    { key: "washing_stop", label: "Конец мойки" },
  ];

  const currentStepIndex = steps.findIndex((step) => step.key === stage);

  const canJog =
    active &&
    !homingActive &&
    !busy &&
    ["set_zero", "material_start", "material_end"].includes(stage ?? "");

  const currentAction = {
    set_zero: {
      label: "Установить ноль",
      description: "Установить текущую позицию как нулевую",
    },
    material_start: {
      label: "Зафиксировать начало материала",
      description: "Сохранить текущую позицию начала материала",
    },
    material_end: {
      label: "Зафиксировать конец материала",
      description: "Сохранить текущую позицию конца материала",
    },
    ready_to_unwind: {
      label: "Начать автоматическую отмотку",
      description: "Отмотать материал до точки начала",
    },
  }[stage ?? "set_zero"];

  const points = [
    {
      key: "zero",
      label: "Нулевая позиция",
      value: 0,
      ready: active && stage !== "set_zero",
    },
    {
      key: "material_start",
      label: "Начало материала",
      value: session?.material_start_turns,
      ready: session?.material_start_turns != null,
    },
    {
      key: "material_end",
      label: "Конец материала",
      value: session?.material_end_turns,
      ready: session?.material_end_turns != null,
    },
    {
      key: "washing_stop",
      label: "Конец мойки",
      value: session?.washing_stop_turns,
      ready: session?.washing_stop_turns != null,
    },
  ];

  async function handleCurrentAction() {
    if (!active || busy || session?.running) return;

    switch (stage) {
      case "set_zero":
        await execute(() => action.mutateAsync("reset-position"));
        break;

      case "material_start":
        await execute(() => mark.mutateAsync("material_start"));
        break;

      case "material_end":
        await execute(() => mark.mutateAsync("material_end"));
        break;

      case "ready_to_unwind":
        setError("Автоматическую отмотку подключим на следующем этапе");
        break;
    }
  }

  return (
    <main className="mx-auto max-w-5xl space-y-6 p-6">
      <div>
        <h1 className="text-3xl font-bold">Калибровка</h1>
        <p className="text-muted-foreground">
          Настройка положения ролика и контрольных точек
        </p>
      </div>

      <Card>
        <CardHeader>
          <CardTitle>Материал</CardTitle>
        </CardHeader>

        <CardContent className="flex flex-wrap items-center gap-3">
          <Select
            value={materialId}
            onValueChange={(value) => setMaterialId(value ?? "")}
            disabled={active || busy}
          >
            <SelectTrigger className="w-64">
              <SelectValue placeholder="Выбери материал" />
            </SelectTrigger>

            <SelectContent>
              {(materialsQuery.data ?? []).map((material) => (
                <SelectItem key={material.id} value={String(material.id)}>
                  {material.name}
                </SelectItem>
              ))}
            </SelectContent>
          </Select>

          <Button
            onClick={handleStart}
            disabled={active || busy || !materialId}
          >
            Начать калибровку
          </Button>

          {active && (
            <span className="text-sm text-muted-foreground">
              Активный материал: {session?.material_profile_id}
            </span>
          )}
        </CardContent>
      </Card>
      <Card>
        <CardHeader>
          <CardTitle>Этапы калибровки</CardTitle>
        </CardHeader>

        <CardContent className="space-y-3">
          {steps.map((step, index) => (
            <div
              key={step.key}
              className={`rounded-lg border p-3 ${
                index === currentStepIndex ? "border-primary bg-primary/5" : ""
              }`}
            >
              <span className="mr-3">{index + 1}.</span>
              {step.label}
              {index < currentStepIndex && (
                <span className="ml-3 text-green-600">✓</span>
              )}
            </div>
          ))}
        </CardContent>
      </Card>
      <Card>
        <CardHeader>
          <CardTitle>Положение ролика</CardTitle>
        </CardHeader>

        <CardContent className="space-y-4">
          <div className="text-4xl font-mono font-semibold">
            {formatTurns(session?.position_turns)}
            <span className="ml-2 text-base font-normal">оборотов</span>
          </div>

          <div className="grid gap-3 sm:grid-cols-3">
            <div>
              <p className="text-sm text-muted-foreground">Состояние</p>
              <p>
                {!active
                  ? "Неактивна"
                  : session?.running
                    ? "Вращение"
                    : session?.paused
                      ? "Пауза"
                      : "Остановлено"}
              </p>
            </div>

            <div>
              <p className="text-sm text-muted-foreground">Направление</p>
              <p>{session?.direction ?? "—"}</p>
            </div>

            <div>
              <p className="text-sm text-muted-foreground">Частота</p>
              <p>{session?.frequency_hz ?? 0} Hz</p>
            </div>
          </div>
          <div className="space-y-3">
            <p className="text-sm text-muted-foreground">
              {currentAction.description}
            </p>

            <Button
              className="w-full"
              size="lg"
              disabled={!active || busy || session?.running}
              onClick={handleCurrentAction}
            >
              {currentAction.label}
            </Button>
          </div>
        </CardContent>
      </Card>

      {active && stage === "set_zero" && (
        <Card>
          <CardHeader>
            <CardTitle>Автоматический поиск нуля</CardTitle>
          </CardHeader>

          <CardContent className="space-y-4">
            <div className="space-y-1 text-sm">
              <p>
                Датчик:{" "}
                {!sensorQuery.data?.connected
                  ? "Нет связи"
                  : sensorQuery.data.triggered
                    ? "Нажат"
                    : "Отпущен"}
              </p>
              <p>
                Поиск: {homeQuery.data?.state ?? "—"}
              </p>
              {homeQuery.data?.error && (
                <p className="text-red-600">
                  {homeQuery.data.error}
                </p>
              )}
            </div>

            <label className="flex items-start gap-3 text-sm">
              <input
                type="checkbox"
                className="mt-1"
                checked={positionConfirmed}
                disabled={homingActive}
                onChange={(event) =>
                  setPositionConfirmed(event.target.checked)
                }
              />
              <span>
                Подтверждаю, что вал вручную установлен в диапазоне
                от 0 до 1 оборота, а микрик отпущен.
              </span>
            </label>

            <Button
              className="w-full"
              disabled={
                busy ||
                homingActive ||
                !!session?.running ||
                !!session?.brush_running ||
                !positionConfirmed ||
                !sensorQuery.data?.connected ||
                !!sensorQuery.data?.triggered
              }
              onClick={() =>
                execute(async () => {
                  await startHome.mutateAsync();
                  setPositionConfirmed(false);
                })
              }
            >
              {homingActive ? "Поиск нуля..." : "Найти ноль (2 Hz)"}
            </Button>

            <p className="text-xs text-muted-foreground">
              Поиск выполняется в обратном направлении.
              Для остановки используй Stop ниже.
            </p>
          </CardContent>
        </Card>
      )}

      <Card>
        <CardHeader>
          <CardTitle>Управление роликом</CardTitle>
        </CardHeader>

        <CardContent className="space-y-4">
          <div className="space-y-3">
            <div className="flex items-center justify-between">
              <span>Скорость ролика</span>
              <span className="font-mono font-semibold">
                {rollerSpeedPercent}%
              </span>
            </div>

            <Slider
              min={1}
              max={100}
              step={1}
              value={[rollerSpeedPercent]}
              onValueChange={(value) =>
                setRollerSpeedPercent(
                  typeof value === "number" ? value : (value[0] ?? 10),
                )
              }
              disabled={!active || busy}
            />
          </div>

          <div className="flex flex-wrap gap-3">
            <Button disabled={!canJog} onClick={() => handleJog("reverse")}>
              ◀ Reverse
            </Button>

            <Button disabled={!canJog} onClick={() => handleJog("forward")}>
              Forward ▶
            </Button>

            <Button
              variant="outline"
              disabled={!active || busy || !session?.running}
              onClick={() => execute(() => action.mutateAsync("pause"))}
            >
              Pause
            </Button>

            <Button
              variant="destructive"
              disabled={!active || (busy && !homingActive)}
              onClick={() => execute(() => action.mutateAsync("stop"))}
            >
              Stop
            </Button>
          </div>
        </CardContent>
      </Card>
      <Card>
        <CardHeader>
          <CardTitle>Управление щётками</CardTitle>
        </CardHeader>

        <CardContent className="space-y-4">
          <div className="space-y-3">
            <div className="flex items-center justify-between">
              <span>Скорость щёток</span>
              <span className="font-mono font-semibold">
                {brushSpeedPercent}%
              </span>
            </div>

            <Slider
              min={0}
              max={100}
              step={1}
              value={[brushSpeedPercent]}
              onValueChange={(value) =>
                setBrushSpeedPercent(
                  typeof value === "number" ? value : (value[0] ?? 30),
                )
              }
              disabled={!active || busy}
            />
          </div>
          <div className="flex flex-wrap gap-3">
            <Button
              disabled={!active || busy || homingActive || brushSpeedPercent <= 0}
              onClick={() =>
                execute(() =>
                  startBrush.mutateAsync(brushPercentToHz(brushSpeedPercent)),
                )
              }
            >
              Запустить щётки
            </Button>

            <Button
              variant="outline"
              disabled={!active || busy || !session?.brush_running}
              onClick={() => execute(() => stopBrush.mutateAsync())}
            >
              Остановить щётки
            </Button>
          </div>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>Контрольные точки</CardTitle>
        </CardHeader>

        <CardContent className="space-y-3">
          {points.map((point) => (
            <div
              key={point.key}
              className="flex items-center justify-between gap-3 rounded-lg border p-4"
            >
              <div className="space-y-1">
                <p className="font-medium">{point.label}</p>

                <p className="font-mono text-sm text-muted-foreground">
                  {point.ready
                    ? `${formatTurns(point.value)} оборотов`
                    : "Не установлена"}
                </p>
              </div>

              <div
                className={
                  point.ready
                    ? "flex items-center gap-2 text-green-600"
                    : "flex items-center gap-2 text-muted-foreground"
                }
              >
                <span
                  className={
                    point.ready
                      ? "h-2.5 w-2.5 rounded-full bg-green-500"
                      : "h-2.5 w-2.5 rounded-full bg-muted-foreground/40"
                  }
                />

                <span className="text-sm font-semibold">
                  {point.ready ? "READY" : "PENDING"}
                </span>
              </div>
            </div>
          ))}
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>Завершение калибровки</CardTitle>
        </CardHeader>

        <CardContent className="flex flex-wrap gap-3">
          <Button disabled={!canComplete} onClick={handleComplete}>
            Сохранить калибровку
          </Button>

          <Button
            variant="outline"
            disabled={!active || busy}
            onClick={handleCancel}
          >
            Отменить
          </Button>
        </CardContent>
      </Card>

      {error && (
        <div
          role="alert"
          className="rounded-lg border border-red-500/30 bg-red-500/10 p-4 text-red-600"
        >
          {error}
        </div>
      )}

      {message && (
        <div className="rounded-lg border p-4 text-sm">{message}</div>
      )}

      {sessionQuery.isError && !active && (
        <p className="text-sm text-muted-foreground">
          Активной сессии калибровки нет.
        </p>
      )}
    </main>
  );
}
